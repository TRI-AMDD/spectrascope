128    def load_file(self, fn:str):
129        """Load data from a file.
130
131        Parameters
132        ----------
133        fn : str
134            The file path.
135        
136        Returns
137        -------
138        data : np.array
139            The loaded data.
140        """
141        
142        _, ext = os.path.splitext(fn)
143        if ext == ".txt":
144            for separator in (" ", ",", "\t", ";"):  # Testing random separators
145                data = pd.read_csv(fn, header=None, sep=separator).values
146                if data.shape[-1] > 1:  # We likely found the separator
147                    break
148        elif ext == ".csv":
149            data = pd.read_csv(fn, header=None, sep=",").values
150        elif ext == ".tsv":
151            data = pd.read_csv(fn, header=None, sep="\t").values
152        elif ext in {".xls", ".xlsx"}:
153            data = pd.read_excel(fn, header=None).values
154        elif ext == ".npy":
155            data = np.load(fn)
156        elif ext == ".json":
157            data = pd.read_json(fn)
158        else:
159            raise ValueError(
160                f"Please provide a filename with extension in {{.txt, .csv, .tsv, .xls, .xlsx, .npy, .json}}. You passed {filename}"
161            )
162        return data.to_numpy()

567        savef=None,
568        use_pipeline=True,
569        top_features=20,
570        scale_X=True,
571        **kwargs,
572    ):
573        """Fit sklearn model or pipeline and save results.
574
575        Parameters
576        ----------
577        Xattr : str
578            X attribute to fit model on.
579        yattr : str
580            y attribute to fit model on.
581        test_size : float or int
582            input for sklearn train_test_split test_size. 
583        savef : str or None, optional, default None
584            filename to save model.
585        use_pipeline : bool, optional, default=True
586            whether to use sklearn pipeline and GridSearchCV.
587        top_features : int, optional, default=20. 
588            number of top features to save for future analysis. 
589        scale_X : bool, optional, default=True. 
590            whether or not to scale X data before fitting using StandardScaler.
591        """
592        self.split_X(Xattr, test_size, scale_X=scale_X)
593        self.split_y(yattr, test_size)
594
595        X_train = getattr(self, Xattr + "_train")
596        X_test = getattr(self, Xattr + "_test")
597
598        y_train = getattr(self, yattr + "_train")
599        y_test = getattr(self, yattr + "_test")
600
601        if use_pipeline:
602            P, score, ypred = self.sk_pipeline(
603                X_train, X_test, y_train, y_test, **kwargs
604            )
605            model_name = P[0].best_estimator_.__str__().split("(")[0]
606        else:
607            P, score, ypred = self.sk(X_train, X_test, y_train, y_test, **kwargs)
608            model_name = P.__str__().split("(")[0]
609
610        if "RandomForest" in model_name:
611            try:
612                feature_importances = P.feature_importances_
613            except AttributeError:
614                feature_importances = P[0].best_estimator_.feature_importances_
615            # choose top features
616            selected_features = np.argsort(feature_importances[::-1])[:top_features]
617            selected_vals = feature_importances[np.argsort(feature_importances[::-1])][
618                :top_features
619            ]
620        elif "LogisticRegression" in model_name:
621            selected_features = np.argsort(P.coef_)[::-1][:top_features]
622            selected_vals = P.coef_[np.argsort(P.coef_)[::-1]][:top_features]
623        else:
624            selected_features = []
625            selected_vals = []
626            # changes dataset: (X[:,sorted(sorted_best_features)])
627
628        fit_dict = {
629            "P": P,
630            "score": score,
631            "ypred": ypred,
632            "selected_features": selected_features,
633            "scaled": scale_X,
634            "selected_vals": selected_vals,
635        }
636        setattr(self, Xattr + "_" + yattr + "_" + model_name, fit_dict)
637
638        perc_err = np.mean(np.abs((y_test - ypred) / y_test)) * 100
639
640        if model_name not in self.best_fits:
641            self.best_fits[model_name] = {Xattr + "_" + yattr: score}  # score
642        else:
643            self.best_fits[model_name][Xattr + "_" + yattr] = score  # score
644
645        if savef is not None:
646            self.check_dir_exists(f"{self.save_path}/{self.today}")
647            with open(
648                f"{self.save_path}/{self.today}/sk_{model_name}_{self.timenow}_{Xattr}_{yattr}_{savef}.p",
649                "wb",
650            ) as f:
651                pickle.dump(P, f)
