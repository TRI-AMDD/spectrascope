324    def av_distance(self, v):
325        distances = np.triu(cdist(v, v)).ravel()
326        nonzero_distances = distances.nonzero()
327        av_dist = distances[nonzero_distances].mean()
328        return av_dist

391        savef=None,
392        use_pipeline=False,

395    ):

397        from sklearn.base import clone

407        #calculating the entropy of the different yattr columns

409        best_order = np.argsort(entropies)
410        yattr_ordered = np.array(yattrs)[best_order]
411
412        sorted_truth = self.ground_truth_df[yattr_ordered]
413        similarity_results = np.zeros([len(yattrs), masks.shape[0], masks.shape[0]])
414        for i, yattr in enumerate(yattr_ordered):
415            selected = yattr_ordered[: i + 1]
416            similarity_result = self.similarity_matrix(
417                mask_df[selected].to_numpy(), sorted_truth.to_numpy()[:, : i + 1]
418            )
419            similarity_results[i] = similarity_result
420
421        return similarity_results, yattr_ordered
