 27    A class that creates feature vectors extracted from 1D array data. 
 28    Current features developed for spectral data, but could be applied to any 1D data.

 50    def pointwise_features(
 51        self, xrange, return_features: bool = False, fill_value="extrapolate"
 52    ):
 53        """Interpolates self.energies, self.intensities onto ne evenly spaced points.
 54
 55        Parameters
 56        ----------
 57            xrange : np.array
 58                range of x values to perform interpolation on. (np.linspace(xmin,xmax))
 59            return_features : bool, default=False
 60                Flag indicating whether to return the features
 61            fill_value:  array-like or (array-like, array_like) or “extrapolate”, optional, default="extrapolate"
 62                fill value for scipy.interp1d for values outside of xrange (see scipy.interpolate.interp1d).
 63        """
 64        
 65        f = interp1d(
 66            self.energies,
 67            self.intensities,
 68            bounds_error=False,
 69            fill_value="extrapolate",
 70        )
 71
 72        self.xinterp = xrange
 73        self.yinterp = f(self.xinterp)
 74        self.de = np.diff(self.xinterp)[0]
 75
 76        if return_features:
 77            return self.yinterp

 79    def generate_bins(
 80        self, N: int, x: np.array = None, y: np.array = None, verbose=False
 81    ):
 82        """Generate N bins for polynomial fitting. 
 83        
 84        Generate N bins even in energy for self.energies. If custom x and y not given, performs binning on
 85        interpolated data if it exists. Otherwise, performs on self.energies, self.intensities.
 86        
 87        Parameters
 88        ----------
 89        N : int 
 90            The number of bins to generate.
 91        x : np.array, optional
 92            Energy values to generate bins for. Default None uses full spectrum.
 93        y : np.array, optional
 94            Intensity values to generate bins for. Default None uses full spectrum.
 95        verbose : bool, optional
 96            print progress. Default False.
 97            
 98        Attributes
 99        ----------
100            bin_indices_{N} : np.array
101                An array of bin indices for the N bins. Shape is (len(x),).
102            bin_indices_{round(x.min())}_{round(x.max())}_{N} : np.array
103                An array of bin indices for N bins over a subset of full energy range. Shape is (len(x),).
104        """
105        
106        if x is None:
107            # use default energies and intensities
108            # always perform functions on interpolated data
109            x, y = self.check_xinterp(verbose)
110
111            use_default_energies = True
112        else:
113            assert len(x) == y.shape[-1], "x and y must be the same length"
114            use_default_energies = False
115
116        bin_width = (max(x) - min(x)) / N
117        bin_indices = np.zeros(len(x), dtype=int)
118        # bin_indices = []
119
120        for i, val in enumerate(x):
121            index = int(((val) - min(x)) / bin_width)
122            bin_indices[i] = index
123            if bin_indices[i] == N:
124                bin_indices[i] -= 1
125
126        # if bin_indices.max() != N-1:
127        #     raise ValueError('Cannot generate even bins with the given N on this range')
128        if len(np.unique(bin_indices)) == len(np.arange(N)):
129            if (np.unique(bin_indices) == np.arange(N)).all():
130
131                if use_default_energies:
132                    setattr(self, f"bin_indices_{N}", bin_indices)
133                    if verbose:
134                        print(
135                            f"Generated {N} bins for energies across the full energy range, attribute: bin_indices_{N}"
136                        )
137                else:
138                    setattr(
139                        self,
140                        f"bin_indices_{round(x.min())}_{round(x.max())}_{N}",
141                        bin_indices,
142                    )
143                    if verbose:
144                        print(
145                            f"Generated {N} bins for energies between {round(x.min(),2)} and {round(x.max(),2)}, attribute: bin_indices_{round(x.min())}_{round(x.max())}_{N}"
146                        )
147
148        else:
149            print("proper number of bins not generated, attribute not created")

152        self,
153        N: int,
154        degree: int = 3,
155        return_features: bool = False,
156        verbose=False,
157        **kwargs,
158    ):
159        """Performs polynomial fitting on N energy equivalent sections.
160
161        Parameters
162        ----------
163        N : int 
164            The number of energy equivalent sections to perform polynomial fitting on.
165        degree : int, optional, default=3
166            The degree of the polynomial to fit.
167        return_features : bool, optional, default=False
168            Flag indicating whether to return the features.
169        verbose : bool, optional, default=False
170            print progress.
171            
172        Attributes
173        ----------
174        polyfeatures_{N}_{degree}deg : np.array
175            An array of polynomial coefficients for the N bins. Shape is (N, degree + 1).
176        """
177        
178        featuresN = np.zeros([self.intensities.shape[0], 1])
179
180        use_default_energies = not "x" in kwargs
181
182        if use_default_energies:
183            if hasattr(self, "xinterp") == False:
184                if verbose:
185                    print(
186                        "generating features on self.energies, self.intensities...are these already interpolated? "
187                    )
188                x = self.energies
189                y = self.intensities
190            else:
191                x = self.xinterp
192                y = self.yinterp
193        else:
194            x = kwargs["x"]
195            y = kwargs["y"]
196
197        for i in range(0, N):
198
199            if use_default_energies:
200                try:
201                    bin_indices = getattr(self, f"bin_indices_{N}")
202                except AttributeError:
203                    self.generate_bins(N, **kwargs)
204                    bin_indices = getattr(self, f"bin_indices_{N}")
205            else:
206                try:
207                    bin_indices = getattr(
208                        self, f"bin_indices_{round(x.min())}_{round(x.max())}_{N}"
209                    )
210                except AttributeError:
211                    self.generate_bins(N, **kwargs)
212                    bin_indices = getattr(
213                        self, f"bin_indices_{round(x.min())}_{round(x.max())}_{N}"
214                    )
215
216            xbin = x[bin_indices == i]
217            ybin = y[:, bin_indices == i]
218            binrange = xbin.max() - xbin.min()
219
220            # centering the x values
221            xbin = xbin - xbin.mean()
222
223            coefs = np.polyfit(xbin, ybin.T, degree)
224            featuresN = np.concatenate((featuresN, coefs.T), axis=1)
225
226        featuresN = featuresN[:, 1:]
227
228        if use_default_energies:
229            setattr(self, f"polyfeatures_{N}_{degree}deg", featuresN)
230            if verbose:
231                print(
232                    f"Generated {N*(degree+1)} polynomial features using a degree {degree} polynomial for energies across the full energy range split into {N} equivalent sections; attribute: polyfeatures_{N}_{degree}deg"
233                )
234        else:
235            setattr(
236                self,
237                f"polyfeatures_{round(x.min())}_{round(x.max())}_{N}_{degree}deg",
238                featuresN,
239            )
240            if verbose:
241                print(
242                    f"Generated {N*(degree+1)} polynomial features using a degree {degree} polynomial for energies between {round(x.min(),2)} and {round(x.max(),2)} split into {N} equivalent sections, attribute: polyfeatures_{round(x.min())}_{round(x.max())}_{N}_{degree}deg"
243                )
244
245        if return_features:
246            return featuresN

362        """Compute cumulative distribution function: probability that f(x) will take on a value less than or equal to x at x.
363

389        """T-distributed Stochastic neighbor embedding 
390        
391        Converts similarities between data points to joint probabilities and tries to minimize

422        self,
423        scales=np.arange(1, 11),
424        wavelet="mexh",
425        verbose=False,
426        return_features: bool = False,
427    ):
428        """
429        Compute continuous wavelet transform on spectrum: uses inner products to measure the similarity between a signal and an analyzing function (wavelet).
430        Compare signal to shifted and compressed or stretched versions of a wavelet. Can choose different wavelets
431
432        Parameters
433        ----------
434        scales : np.array, optional, Default= np.arange(1,11).
435            Array of ints. 
436        wavelet : pywt.ContinuousWavelet, optional
437            Mother wavelet to compare spectrum to. Default 'mexh'.
438        verbose : bool, optional
439            print progress. Default False.
440        return_features : bool, optional, Default=False. 
441            return features. Returns coefs raveled to len(widths) * len(intensity) to give 1D array for each sample returns shape (n_samples, len(widths)*len(intensity)) for SpectraScope class.
442
443        Attributes
444        ----------
445        cwt_coefs : np.array
446            continuous wavelet transform of the imput signal for the given scales and wavelet. First axis corresponds to the scales, remaining axes match the shape of intensity.
447        cwt_frequencies : np.array
448            if the unit of sampling period are seconds and given, then frequencies are in herts, otherwise a sampling period of 1 assumed.
449        """
450        
451        x, y = self.check_xinterp(verbose)
452        
453        scales = np.array(scales).astype(int)
454        
455        temp_coefs, self.cwt_frequencies = pywt.cwt(y, scales, wavelet, axis=1)
456        cwt_shape = temp_coefs.shape
457
458        # TODO : better way to reshape without for loop
459        self.cwt_coefs = np.zeros([cwt_shape[1], cwt_shape[0], cwt_shape[-1]])
460        for i in np.arange(cwt_shape[1]):
461            self.cwt_coefs[i] = temp_coefs[:, i]
462
463        if return_features:
464            return self.cwt_coefs.reshape(cwt_shape[1], cwt_shape[0] * cwt_shape[2])

467        """
468        Fit n=num_peaks gaussians to intensity.
469        Not vectorized yet. 
470        
471        Parameters
472        ----------

474            Number of gaussians to fit.
475        verbose : bool, optional
476            print progress. Default False.
477        return_features : bool, optional
478            return features. Default True.
479
480        Attributes

484        """

486        self.peak_features = np.array([])

488        # TODO: vectorize
489

491            idxes = [int(i) for i in np.linspace(0, len(x) - 1, num_peaks + 1)]

493                int((idxes[i] + idxes[i + 1]) / 2) for i in range(len(idxes) - 1)
